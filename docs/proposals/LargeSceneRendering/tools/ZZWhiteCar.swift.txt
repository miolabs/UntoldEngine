// Scratch, not part of the engine: a white parked car close up, at a forced level, with
// and without its dark parts (the interior), to tell what the dark patches are.
import simd
@testable import UntoldEngine
import XCTest

final class ZZWhiteCar: BaseRenderSetup {
    override func initializeAssets() {}
    private func waitForGPU() { renderInfo.lastCommandBuffer?.waitUntilCompleted() }
    private func draw(_ n: Int) { for _ in 0 ..< n { renderer.draw(in: renderer.metalView); waitForGPU() } }
    private func save(_ name: String, in dir: URL) {
        if let texture = textureResources.antiAliasingTexture ?? textureResources.colorMap, let image = textureToCGImage(texture: texture) {
            saveResultToDisk(image, named: name, in: dir)
        }
    }

    func testWhiteCar() async throws {
        let env = ProcessInfo.processInfo.environment
        guard let pack = env["BIM_PACK"], let out = env["BIM_OUT"] else { throw XCTSkip("BIM_PACK/BIM_OUT not set") }
        let outputDir = URL(fileURLWithPath: out)
        try FileManager.default.createDirectory(at: outputDir, withIntermediateDirectories: true)
        LoadingSystem.shared.resourceURLFn = getResourceURL
        applyIBL = false; renderEnvironment = false; ambientIntensity = 0.6
        let rootId = createEntity()
        let loaded = expectation(description: "pack")
        setEntityMeshAsync(entityId: rootId, filename: pack, withExtension: "untoldpack") { _ in loaded.fulfill() }
        await fulfillment(of: [loaded], timeout: 3600)
        traverseSceneGraph()
        let all = scene.getAllEntities()
        let camera = createEntity(); createGameCamera(entityId: camera); CameraSystem.shared.activeCamera = camera
        far = 2000

        // The parked cars and their parts; a car is white when its largest part's material is light.
        struct Car { var root: EntityID; var parts: [EntityID]; var lods: [LODComponent]; var centre: SIMD3<Float>; var radius: Float; var paintLuminance: Float; var dark: [EntityID] }
        var cars: [Car] = []
        for placement in all where getEntityName(entityId: placement).lowercased().contains("aparcado") && !hasComponent(entityId: placement, componentType: RenderComponent.self) {
            var parts: [EntityID] = []; var lods: [LODComponent] = []; var dark: [EntityID] = []
            var lower = simd_float3(repeating: .greatestFiniteMagnitude), upper = -lower
            var largestArea: Float = 0; var paint: Float = 0
            var queue = [placement]
            while let e = queue.popLast() {
                if let render = scene.get(component: RenderComponent.self, for: e) {
                    parts.append(e)
                    if let lod = scene.get(component: LODComponent.self, for: e) { lods.append(lod) }
                    if let local = scene.get(component: LocalTransformComponent.self, for: e), let world = scene.get(component: WorldTransformComponent.self, for: e) {
                        let (a, b) = worldAABB_MinMax(localMin: local.boundingBox.min, localMax: local.boundingBox.max, worldMatrix: world.space)
                        lower = simd_min(lower, a); upper = simd_max(upper, b)
                        let size = b - a; let area = size.x * size.y + size.y * size.z + size.x * size.z
                        var lum: Float = 0; var count: Float = 0
                        for mesh in render.mesh { for sub in mesh.submeshes { if let m = sub.material { let c = m.baseColorValue; lum += 0.3 * c.x + 0.59 * c.y + 0.11 * c.z; count += 1 } } }
                        let meanLum = count > 0 ? lum / count : 0
                        if area > largestArea { largestArea = area; paint = meanLum }
                        if meanLum < 0.25 { dark.append(e) }
                    }
                }
                queue.append(contentsOf: getEntityChildren(parentId: e))
            }
            if lower.x < upper.x { cars.append(Car(root: placement, parts: parts, lods: lods, centre: (lower + upper) * 0.5, radius: simd_length(upper - lower) * 0.5, paintLuminance: paint, dark: dark)) }
        }
        print("WHITECAR cars=\(cars.count) luminances=\(cars.prefix(12).map { String(format: "%.2f", $0.paintLuminance) })")
        guard let car = cars.max(by: { $0.paintLuminance < $1.paintLuminance }) else { return }
        print("WHITECAR chosen \(getEntityName(entityId: car.root)) luminance=\(car.paintLuminance) parts=\(car.parts.count) dark=\(car.dark.count)")
        let eye = car.centre + simd_normalize(simd_float3(0.3, 0.45, 1)) * car.radius * 1.5
        cameraLookAt(entityId: camera, eye: eye, target: car.centre, up: simd_float3(0, 1, 0))
        // KEEP_MESHES / HIDE_MESHES: comma-separated lowercase substrings of part names.
        let keep = (env["KEEP_MESHES"] ?? "").split(separator: ",").map { String($0).lowercased() }.filter { !$0.isEmpty }
        let hide = (env["HIDE_MESHES"] ?? "").split(separator: ",").map { String($0).lowercased() }.filter { !$0.isEmpty }
        var hiddenNames: [String] = []
        for e in car.parts {
            let name = getEntityName(entityId: e).lowercased()
            let keepIt = keep.isEmpty || keep.contains { name.contains($0) }
            let hideIt = hide.contains { name.contains($0) }
            if !keepIt || hideIt { scene.get(component: RenderComponent.self, for: e)?.isVisible = false; hiddenNames.append(name) }
        }
        if !hiddenNames.isEmpty { print("WHITECAR hidden \(hiddenNames.count) parts: \(hiddenNames.sorted().prefix(40))") }
        let levels = env["ONLY_LEVEL"].flatMap { Int($0) }.map { [$0] } ?? [0, 3]
        let darkVariants = env["NO_NODARK"] != nil ? [false] : [false, true]
        for level in levels {
            for lod in car.lods { lod.forcedLOD = level }
            LODSystem.shared.reset()
            for hideDark in darkVariants {
                if hideDark { for e in car.dark { scene.get(component: RenderComponent.self, for: e)?.isVisible = false } }
                draw(10)
                save("white_lod\(level)\(hideDark ? "_nodark" : "")\(env["TAG"] ?? "")", in: outputDir)
            }
        }
    }
}
